// Firestore Security Rules for Waitlist Collection
// Copy these rules to Firebase Console > Firestore Database > Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Allow anyone to write to waitlist collection (for landing page)
    match /waitlist/{document} {
      allow create: if true;  // Anyone can add to waitlist
      allow read, update, delete: if false;  // Only admins can read/modify
    }
    
    // Keep your existing rules for other collections
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
